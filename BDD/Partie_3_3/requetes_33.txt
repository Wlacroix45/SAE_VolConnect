% Domaine des heures (lié aux faits succ)
heure(X) :- succ(X, _).
heure(X) :- succ(_, X).

% <= : base + transitivité
avant_ou_egal(X, Y) :- succ(X, Y).
avant_ou_egal(X, Z) :- succ(X, Y), avant_ou_egal(Y, Z).

% Réflexivité sûre (X est lié par heure(X))
avant_ou_egal(X, X) :- heure(X).

trajet(VilleDep, VilleArr, HDep, HArr) :-
vol(Id, Nom, HDep, HArr, Comp, TermDep, TermArr, VilleDep, VilleArr, PaysDep, PaysArr).

trajet(VilleDep, VilleArr, HDep, HArr) :-
vol(Id1, Nom1, HDep1, HArr1, Comp1, TermDep1, TermArr1, VilleDep, VilleInter, PaysDep1, PaysInter1),
trajet(VilleInter, VilleArr, HDep2, HArr2),
avant_ou_egal(HArr1, HDep2),
HDep = HDep1,
HArr = HArr2,
VilleDep != VilleArr.

ville_connectee(VilleDep, VilleArr, HDep, HArr) :-
trajet(VilleDep, VilleArr, HDep, HArr),
VilleDep != VilleArr.
